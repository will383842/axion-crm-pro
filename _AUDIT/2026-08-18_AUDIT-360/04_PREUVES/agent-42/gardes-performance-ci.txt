    name: axe-core Playwright
    runs-on: ubuntu-24.04
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { version: 9 }
      - uses: actions/setup-node@v7
        with: { node-version: '22', cache: 'pnpm', cache-dependency-path: frontend/pnpm-lock.yaml }
      - name: Install deps
        working-directory: frontend
        run: pnpm install --frozen-lockfile || pnpm install
      - name: Install Playwright browsers
        working-directory: frontend
        run: pnpm exec playwright install --with-deps chromium
      # La cible doit EXISTER sur le runner. `http://app:5173` était le nom de
      # service du docker-compose local : sur un runner GitHub il ne résout pas
      # (`net::ERR_NAME_NOT_RESOLVED`), et `webServer` est neutralisé quand
      # `CI=true`. Depuis la création de ce job, les 4 pages échouaient donc
      # AVANT le premier contrôle axe : la porte n'a jamais mesuré une seule
      # règle d'accessibilité, elle a seulement toujours rougi.
      # On construit l'application et on la sert réellement ; c'est
      # `playwright.config.ts` qui démarre et arrête `vite preview`.
      - name: Build frontend (la cible que la porte doit mesurer)
        working-directory: frontend
        run: pnpm build
      - name: Run a11y E2E
        working-directory: frontend
